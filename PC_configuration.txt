# Link configuration

ip link add link eth1 name eth1.10 type vlan id 10

# Link up

ip link set eth1.10 up 


# Assing IP static

ip addr add 172.9.10.1/24 dev eth1.10

# Verify 

ip addr show eth1.10

# Gateway configuration

ip route add default via 172.9.10.254

# if your default route is different, please delete and add 172.9.10.254

ip route del default
